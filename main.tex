\begin{frame}{What is in the current GSGD release?}
  \vspace*{0.40cm}
  \centering
  \stat{2.60cm}{60,538}{gauge metadata records}\hspace{0.35cm}
  \stat{2.60cm}{51,633}{matched time-series files}\hspace{0.35cm}
  \stat{2.60cm}{159}{countries and regions}\hspace{0.35cm}
  \stat{2.60cm}{34}{contributing data sources}

  \vspace{0.75cm}
  \begin{minipage}{0.89\textwidth}
    \small
    GSGD links harmonized station metadata to traceable gauge-level observations
    collected from national agencies, APIs, research archives, and global catalogues.
    The gauge is the unit of organization: every retained record keeps a persistent
    identifier, source provenance, station-type status, and file relationship.
  \end{minipage}
\end{frame}

  \vspace{0.34cm}
  {\bfseries\Large\color{GSGDRed}GSGD adds 29,579 metadata records, a 95.5\% increase.}\par

  \vspace{0.42cm}
  \begin{minipage}{0.90\textwidth}
    \small
    This is a coverage comparison, not a claim that every source is freely
    redistributable. GSGD selects one primary station inventory for each country;
    alternative catalogues remain visible but are not added to the selected total.
  \end{minipage}
\end{frame}

    \column{0.23\textwidth}
      \centering
      {\sffamily\scriptsize\color{GSGDMuted}PROBABLE RIVER}\par
      {\bfseries\Huge 31,559}\par
      \small consistent supporting evidence
    \column{0.20\textwidth}
      \centering
      {\sffamily\scriptsize\color{GSGDMuted}REVIEW REQUIRED}\par
      {\bfseries\Huge\color{GSGDGold}380}\par

    \column{0.20\textwidth}
      \centering
      {\sffamily\scriptsize\color{GSGDMuted}NON-RIVER}\par
      {\bfseries\Huge\color{GSGDRed}525}\par
      \small facility type identified
  \end{columns}

  \vspace{0.78cm}
  \centering
  \begin{minipage}{0.88\textwidth}
    \small
    \texttt{review\_required} means the available evidence is incomplete or conflicting.
    \texttt{non\_river} means the source or documented assessment identifies a canal,
    reservoir, groundwater well, spring, diversion, or another non-river facility.
  \end{minipage}
\end{frame}

% 6
\begin{frame}{Station type is decided from source evidence first}
  \vspace*{0.04cm}
  \footnotesize
  \renewcommand{\arraystretch}{0.94}
  \begin{tabularx}{\linewidth}{@{}>{\centering\arraybackslash}p{0.75cm}p{4.45cm}X@{}}
    \toprule
    Priority & Evidence & How it is used \\
    \midrule
    1 & Official station type or measured parameter & authoritative classification \\
    2 & Station name, river name, and description & identify natural and engineered facilities \\
    3 & Source-specific catalogue definitions & interpret agency codes in their own context \\
    4 & HarP river network within 1 km & primary spatial support for ambiguous records \\
    5 & MERIT Hydro and JRC surface water & fallback drainage and surface-water context \\
    \bottomrule
  \end{tabularx}

  \vspace{0.26cm}
  {\color{GSGDRed}\rule{\linewidth}{0.7pt}}\par
  \vspace{0.12cm}
  \small
  \textbf{Spatial evidence supports assessment; it does not make the final decision.}
  Agreement can support \texttt{probable\_river}. A spatial mismatch sends the record
  to review and never overrides an authoritative agency classification.
\end{frame}

\begin{frame}{The current product links 51,633 gauges to time series}
  \vspace*{0.02cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.38\textwidth}
      \vspace{0pt}
      {\sffamily\scriptsize\color{GSGDMuted}TIME-SERIES INVENTORY}\par
      {\bfseries\Huge\color{GSGDRed}51,633}\par
      \vspace{0.16cm}
      {\large 85.3\% of all metadata records}\par
      \vspace{0.40cm}
      {\bfseries\Large 33} countries currently contribute held series
    \column{0.56\textwidth}
      \vspace{0pt}
      \small
      \begin{itemize}
        \item Metadata and physical station files are matched by persistent gauge code.
        \item Every packaged file has start and end dates; no metadata path is missing
          its file, and no physical file lacks metadata.
        \item The 521 matched non-river series are retained with their facility flags.
        \item GRDC/EWA observations and other licence-restricted holdings are treated
          according to source-specific redistribution conditions.
      \end{itemize}
  \end{columns}

  \vspace{0.24cm}
  \centering
  \begin{minipage}{0.88\textwidth}
    \small\bfseries
    The practical result is one traceable product: broader coverage than GSIM,
    explicit station-type uncertainty, and a one-to-one link from metadata to each
    held time-series file.
  \end{minipage}
\end{frame}
